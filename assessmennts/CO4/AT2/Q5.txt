import tensorflow as tf
from tensorflow.keras.applications import InceptionV3
from tensorflow.keras.applications.inception_v3 import preprocess_input
from tensorflow.keras.layers import Input, Dense, LSTM, Embedding, Dropout, RepeatVector, TimeDistributed
from tensorflow.keras.models import Model

IMG_SIZE = (299, 299)
VOCAB_SIZE = 10000
EMBEDDING_DIM = 256
LSTM_UNITS = 512
MAX_LENGTH = 30
NUM_FRAMES = 20

cnn = InceptionV3(
    weights="imagenet",
    include_top=False,
    pooling="avg"
)

for layer in cnn.layers:
    layer.trainable = False

video_input = Input(shape=(NUM_FRAMES, 299, 299, 3))

features = TimeDistributed(cnn)(video_input)
features = TimeDistributed(
    Dense(EMBEDDING_DIM, activation="relu")
)(features)

encoder = LSTM(
    LSTM_UNITS,
    return_state=True
)

_, state_h, state_c = encoder(features)

caption_input = Input(shape=(MAX_LENGTH,))

caption_embedding = Embedding(
    VOCAB_SIZE,
    EMBEDDING_DIM,
    mask_zero=True
)(caption_input)

decoder = LSTM(
    LSTM_UNITS,
    return_sequences=True
)

decoder_output = decoder(
    caption_embedding,
    initial_state=[state_h, state_c]
)

decoder_output = Dropout(0.3)(decoder_output)

outputs = TimeDistributed(
    Dense(VOCAB_SIZE, activation="softmax")
)(decoder_output)

model = Model(
    [video_input, caption_input],
    outputs
)

model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

model.summary()
